77template<typename Container>
78void ivedimas_is_failo(Container &S, int meniu_pasirinkimas) {
79 string path;
80 cout << "Iveskite failo, is kurio norite nuskaityti, pavadinima (pvz., kursiokai.txt): ";
81 cin >> path;
82 if (path.empty() || path.size() < 4 || path.substr(path.size() - 4) != ".txt") {
83 cin.clear();
84 throw std::runtime_error("Klaida! Failo pavadinimas turi baigtis .txt...\n");
85 }
86 std::ifstream in(path);
87 if (!in) {
88 throw std::runtime_error("Klaida! Nepavyko atidaryti failo...\n");
89 }
90 string eil;
91 getline(in, eil); // skip header
92 while (getline(in, eil)) {
93 int paz_suma = 0;
94 if (eil.empty()) continue;
95
96 std::istringstream ss(eil);
97 int paz;
98 Studentas tmp;
99 string vrd, pvrd;
100 ss >> vrd >> pvrd;
101 tmp.setVardas(vrd);
102 tmp.setPavarde(pvrd);
103 if (meniu_pasirinkimas == 1) {
104 while (ss >> paz) {
105 if (paz > 0 && paz <= 10) {
106 tmp.getNdRez().push_back(paz);
107 paz_suma += paz;
108 } else {
109 throw std::runtime_error("Klaida! Pazymiai turi buti desimtbaleje sistemoje...\n");
110 }
111 }
112 }
113 if (!tmp.getNdRez().empty()) {
114 tmp.setEgzRez(tmp.getNdRez().back());
115 tmp.getNdRez().pop_back();
116 } else if (meniu_pasirinkimas == 1) {
117 throw std::runtime_error("Klaida! Mokinio " + tmp.getVardas() + " "
118 + tmp.getPavarde() + " namu darbu pazymiai nerasti...\n");
119 }
120 tmp.setVidurkis(static_cast<double>(paz_suma - tmp.getEgzRez())
121 / static_cast<double>(tmp.getNdRez().size()));
122 S.push_back(tmp);
123 }
124}
