49 pazSuma += paz;

53 tmp.setEgzRez(tmp.getNdRez().back());
54 tmp.getNdRez().pop_back();
55 }
56 tmp.setVidurkis(static_cast<double>(pazSuma - tmp.getEgzRez())
57 / static_cast<double>(tmp.getNdRez().size()));
58 S.push_back(tmp);

60 in.close();
